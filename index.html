async function sendModuleChat(moduleType) { const input = document.getElementById(`input-${moduleType}`); const text = input.value.trim(); if(!text) return; const box = document.getElementById(`chat-messages-${moduleType}`); box.innerHTML += `
${text}
`; input.value = ''; box.scrollTop = box.scrollHeight; const loadingId = 'loading-' + Date.now(); box.innerHTML += `
Altra AI is processing your request...
`; box.scrollTop = box.scrollHeight; try { const response = await fetch('/api/chat', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ prompt: text, clientId: userProfile.name, service: moduleType, country: document.getElementById('biz-country').value, currency: "₦", category: document.getElementById('biz-category').value }) }); const data = await response.json(); document.getElementById(loadingId).remove(); if (data.success) { box.innerHTML += `
${data.reply}
`; } else { throw new Error(data.error || 'API response failed'); } } catch (error) { document.getElementById(loadingId).remove(); box.innerHTML += `
Error connecting to server: ${error.message}
`; } box.scrollTop = box.scrollHeight; }
